"""Does CheckoutProducer.send report success for an event Kafka rejected? (A topic that doesn't exist; auto-create is
off, so the broker refuses delivery asynchronously.) Prints evidence only."""

import sys
import time
import uuid
from datetime import date
from pathlib import Path

ROOT = Path(r"C:\projects\project1\points-posting-investigator")
sys.path.insert(0, str(ROOT))

from confluent_kafka import Producer  # noqa: E402
from confluent_kafka.schema_registry import SchemaRegistryClient  # noqa: E402

from app.pipeline import schemas, topics  # noqa: E402
from app.pipeline.producer import CheckoutProducer, build_event  # noqa: E402

registry = SchemaRegistryClient({"url": "http://127.0.0.1:8081", "timeout": 10})
missing = f"diag.missing.{uuid.uuid4().hex[:8]}"
reports = []
raw = Producer(
    {
        "bootstrap.servers": "127.0.0.1:9092",
        "enable.idempotence": True,
        "acks": "all",
        "topic.metadata.propagation.max.ms": 1000,
        "message.timeout.ms": 5000,
    }
)
producer = CheckoutProducer(raw, registry, missing, schema=schemas.latest(topics.CHECKED_OUT))
record = build_event(member_number="GR-00001", stay_id="BR-0001", check_in=date(2026, 11, 14), nights=2, room_rate_cents=1)
started = time.monotonic()
try:
    sent = producer.send(record)
    outcome = f"send() RETURNED NORMALLY (event_id {sent.event_id[:8]}...)"
except Exception as exc:
    outcome = f"send() raised {type(exc).__name__}: {exc}"
print(f"{outcome} after {time.monotonic() - started:.1f} s")

# What Kafka itself reported for an identical message, with a delivery callback:
raw.produce(missing, key=b"GR-00001", value=b"x", on_delivery=lambda err, msg: reports.append(err))
raw.flush(15)
print(f"delivery report for the same kind of message: {reports[0] if reports else 'none'}")
